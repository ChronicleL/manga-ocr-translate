#   pip install -r requirements-studio.txt
#   pip install --no-deps PySide6-Fluent-Widgets PySideSix-Frameless-Window
#

# "PySide6" meta package, which drags in PySide6-Addons (QtWebEngine & co., several

# packages are installed with --no-deps and their small real dependencies are
# listed here.
